def bootstrap_dependencies():
    """在程式啟動前，自動檢查並補齊缺少的第三方套件"""

        "plyer": "plyer",
        "win32com": "pywin32",

            missing_packages.append(pip_name)

def parse_time_to_seconds(time_str):
    time_str = time_str.strip().lower()
    if not time_str:
        return 0
    if time_str.isdigit():
        return int(time_str)
    
    total_seconds = 0
    hours = re.search(r'(\d+)\s*h', time_str)
    minutes = re.search(r'(\d+)\s*m', time_str)
    seconds = re.search(r'(\d+)\s*s', time_str)
    
    if hours:   total_seconds += int(hours.group(1)) * 3600
    if minutes: total_seconds += int(minutes.group(1)) * 60
    if seconds: total_seconds += int(seconds.group(1))
    
    if not (hours or minutes or seconds):
        return None
    return total_seconds

def scan_installed_apps():
    apps = {}
    reg_paths = [
        (winreg.HKEY_LOCAL_MACHINE, r"SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall"),
        (winreg.HKEY_LOCAL_MACHINE, r"SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Uninstall"),
        (winreg.HKEY_CURRENT_USER, r"SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall")
    ]
    
    exclude_keywords = [
        "kb", "microsoft", "windows", "security update", "redistributable", 
        "driver", "intel", "amd", "nvidia", "framework", "net framework"
    ]

    for root, path in reg_paths:
        try:
            key = winreg.OpenKey(root, path)
            for i in range(winreg.QueryInfoKey(key)[0]):
                try:
                    sub_key_name = winreg.EnumKey(key, i)
                    sub_key = winreg.OpenKey(key, sub_key_name)
                    
                    try: name, _ = winreg.QueryValueEx(sub_key, "DisplayName")
                    except: continue
                    try: uninstall_str, _ = winreg.QueryValueEx(sub_key, "UninstallString")
                    except: continue
                    
                    try: install_loc, _ = winreg.QueryValueEx(sub_key, "InstallLocation")
                    except: install_loc = ""
                    try: publisher, _ = winreg.QueryValueEx(sub_key, "Publisher")
                    except: publisher = ""

                    try: is_system, _ = winreg.QueryValueEx(sub_key, "SystemComponent")
                    except: is_system = 0
                    try: parent, _ = winreg.QueryValueEx(sub_key, "ParentKeyName")
                    except: parent = ""

                    name_lower = name.lower()
                    pub_lower = publisher.lower()
                    
                    if is_system == 1 or parent != "": continue
                    if any(kw in name_lower for kw in exclude_keywords if kw != "microsoft" or "office" not in name_lower): continue
                    if any(kw in pub_lower for kw in ["microsoft corporation", "intel", "nvidia", "advanced micro devices"]): continue
                    if name.strip() == "" or name in apps: continue
                    
                    apps[name] = {
                        "uninstall_string": uninstall_str,
                        "install_location": install_loc,
                        "key_name": sub_key_name,
                        "publisher": publisher
                    }
                except: continue
            winreg.CloseKey(key)
        except: continue
    return sorted(apps.items(), key=lambda x: x[0].lower())

def deep_clean_leftovers(app_name, info):
    print(f"\n[🧹 深度清理] 正在全力搜捕 [{app_name}] 的殘留垃圾...")
    time.sleep(1)
    
    cleaned_files_count = 0
    cleaned_reg_count = 0
    
    if info["install_location"] and os.path.exists(info["install_location"]):
        try:
            shutil.rmtree(info["install_location"])
            print(f"  -> 已強制抹除原安裝資料夾: {info['install_location']}")
            cleaned_files_count += 1
        except: pass

    sanitize_name = re.sub(r'[^a-zA-Z0-9\s]', '', app_name).split()[0] if app_name.split() else "NONAME"
    search_targets = [sanitize_name]
    if info["publisher"]:
        pub_sanitize = re.sub(r'[^a-zA-Z0-9\s]', '', info["publisher"]).split()[0]
        if len(pub_sanitize) > 3: search_targets.append(pub_sanitize)

    common_paths = [
        os.environ.get("APPDATA"),
        os.environ.get("LOCALAPPDATA"),
        os.environ.get("ProgramFiles"),
        os.environ.get("ProgramFiles(x86)"),
        os.path.join(os.environ.get("SystemDrive", "C:"), "ProgramData")
    ]

    for base_path in common_paths:
        if not base_path or not os.path.exists(base_path): continue
        try:
            for folder in os.listdir(base_path):
                folder_lower = folder.lower()
                if any(target.lower() in folder_lower for target in search_targets if len(target) > 2):
                    full_path = os.path.join(base_path, folder)
                    if os.path.isdir(full_path):
                        try:
                            shutil.rmtree(full_path)
                            print(f"  -> 發現並粉碎 AppData/ProgramData 殘留目錄: {folder}")
                            cleaned_files_count += 1
                        except: pass
        except: pass

    reg_roots = [winreg.HKEY_CURRENT_USER, winreg.HKEY_LOCAL_MACHINE]
    software_paths = [r"Software", r"Software\Wow6432Node"]
    for root in reg_roots:
        for s_path in software_paths:
            try:
                key = winreg.OpenKey(root, s_path, 0, winreg.KEY_ALL_ACCESS)
                for target in search_targets:
                    if len(target) < 3: continue
                    try:
                        sub_key = winreg.OpenKey(key, target, 0, winreg.KEY_ALL_ACCESS)
                        winreg.CloseKey(sub_key)
                        winreg.DeleteKey(key, target)
                        print(f"  -> 已清除登錄檔殘留節點: {s_path}\\{target}")
                        cleaned_reg_count += 1
                    except: pass
                winreg.CloseKey(key)
            except: pass

    print("\n====================================================")
    print(f" 🎉 [{app_name}] 垃圾清理完畢！")
    print(f" 🌟 共清除殘留檔案/目錄: {cleaned_files_count} 處")
    print(f" 🌟 共註銷殘留登錄檔節點: {cleaned_reg_count} 個")
    print("====================================================")

def _async_eject_cd(target_cd):
    try:
        winmm = ctypes.windll.winmm
        winmm.mciSendStringW(f"open {target_cd} type cdaudio alias cd_worker", None, 0, 0)
        result = winmm.mciSendStringW("set cd_worker door open", None, 0, 0)
        winmm.mciSendStringW("close cd_worker", None, 0, 0)
        if result == 0:
            notification.notify(title="光碟機管理", message=f"光碟機 [{target_cd}\\] 託盤已成功彈出！", timeout=5)
        else:
            notification.notify(title="光碟機管理", message=f"光碟機 [{target_cd}\\] 彈出失敗，硬體無響應。", timeout=5)
    except:
        pass

def get_current_windows_version():
    try:
        # 改用 PowerShell 精準提取 Win11 的真實名稱，徹底解決核心誤判問題
        ps_cmd = "powershell -NoProfile -Command \"(Get-ComputerInfo).OsName + ' ' + (Get-ComputerInfo).OsDisplayVersion\""
        res = subprocess.run(ps_cmd, capture_output=True, text=True, errors="ignore", shell=True)
        out = res.stdout.strip()
        if "Windows 11" in out or "Windows 10" in out:
            return out.replace("Microsoft ", "")
        
        import winreg
        key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, r"SOFTWARE\Microsoft\Windows NT\CurrentVersion")
        display_version, _ = winreg.QueryValueEx(key, "DisplayVersion")
        winreg.CloseKey(key)
        
        if os.path.exists(r"C:\Windows\Web\Wallpaper\Windows\img0.jpg"):
            return f"Windows 11 Pro for Workstations {display_version}"
        return f"Windows 10 Pro for Workstations {display_version}"
    except:
        return "Windows 11 Pro"

def convert_edition(saved_password):
    """【功能 8】版本無損轉換核心（含重名專業版過濾機制）"""
    if not verify_password(saved_password): return

    os.system('cls')
    print("==================== Windows 版本無損轉換 ====================")
    current_version_str = get_current_windows_version()
    print(f" 當前系統狀態: {current_version_str}")
    
    print(" 正在讀取微軟核心架構，自動分析本機可升級/轉換的目標版本清單...")
    
    cmd = "dism /online /get-targeteditions"
    res = subprocess.run(cmd, capture_output=True, text=True, errors="ignore", shell=True)
    available_editions = []
    
    for line in res.stdout.split("\n"):
        if "Target Edition :" in line or "目標版本 :" in line:
            edition_id = line.split(":")[-1].strip()
            if edition_id and edition_id not in available_editions: 
                available_editions.append(edition_id)
                
    if not available_editions:
        print("\n[提示] 系統核心回傳：目前您的 Windows 版本已經是最高階，或不支援線上直接轉換。")
        input("\n按 Enter 鍵返回主選單...")
        return

    print("\n[分析完成] 偵測到您的電腦相容並允許無損轉換至以下版本 (已自動過濾重複項):")
    
    displayed_names = set() 
    valid_idx = 1
    filtered_editions = []  

    for edition in available_editions:
        low_ed = edition.lower()
        
        if "professionalworkstation" in low_ed: friendly_name = "Windows 11 專業工作站版 (Pro for Workstations)"
        elif "professional" in low_ed: friendly_name = "Windows 11 專業版 (Professional)"
        elif "iotenterprise" in low_ed: friendly_name = "Windows 11 IoT物聯網企業版 (IoT Enterprise)"
        elif "enterprise" in low_ed: friendly_name = "Windows 11 企業版 (Enterprise)"
        elif "education" in low_ed: friendly_name = "Windows 11 教育版 (Education)"
        elif "cloud" in low_ed: friendly_name = "Windows 11 雲端輕量版 (Cloud Edition)"
        elif "server" in low_ed: continue 
        else: friendly_name = f"Windows 11 {edition}"
        
        # 安全機制：在此攔截並揉碎所有重複出現的 Professional SKU 標記
        if friendly_name in displayed_names:
            continue
            
        displayed_names.add(friendly_name)
        filtered_editions.append(edition)
        print(f" [{valid_idx}] {friendly_name}")
        valid_idx += 1
        
    choice = input("\n請輸入您要轉換的目標編號 (或直接按 Enter 取消): ").strip()
    if choice == '': return
    try:
        choice_idx = int(choice) - 1
        if 0 <= choice_idx < len(filtered_editions):
            target_edition = filtered_editions[choice_idx]
            
            os.system('cls')
            print("\n" * 2)
            print("====================================================")
            print(" ⚠️ 【版本無損轉換安全確認】")
            print("====================================================")
            print(f" 您的系統依然會保持為「Windows 11」，僅變更功能授權階級。")
            print(f" 即將無損轉換至: {target_edition}")
            print(" 轉換過程中系統會自動配置對應組件，請保持網路暢通。")
            print("====================================================")
            confirm = input("確定要立刻執行轉換嗎？[Y/N]: ").strip().upper()
            
            if confirm != 'Y': 
                print("\n[操作取消] 安全退出。")
                time.sleep(1.5)
                return
                
            print(f"\n正在向核心註冊並無損轉換至 [{target_edition}]... 請勿關閉電腦。")
            os.system(f"dism /online /set-edition:{target_edition} /accepteula /norestart")
            print("\n🎉 轉換指令發送成功！微軟組件部署完成後，請自行重啟電腦生效。")
    except Exception as e:
        print(f"[❌ 錯誤] 執行轉換時發生例外: {e}")
    input("\n按 Enter 鍵返回主選單...")
